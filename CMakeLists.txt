cmake_minimum_required(VERSION 3.21)
project(ComplexMinesweeper VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
qt_add_executable(complex_minesweeper WIN32 frontend/main.cpp minesweeping.cpp)
target_include_directories(complex_minesweeper PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(complex_minesweeper PRIVATE Qt6::Widgets)
if(MSVC)
    target_compile_options(complex_minesweeper PRIVATE /utf-8 /W4)
else()
    target_compile_options(complex_minesweeper PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS complex_minesweeper RUNTIME DESTINATION .)
if(WIN32)
    if(Qt6_VERSION VERSION_LESS 6.5)
        qt_generate_deploy_app_script(TARGET complex_minesweeper FILENAME_VARIABLE deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    else()
        qt_generate_deploy_app_script(TARGET complex_minesweeper OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    endif()
    install(SCRIPT ${deploy_script})
endif()
